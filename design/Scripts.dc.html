// чеки за текущий месяц, сгруппированные по продавцу
const since = sema.date('начало месяца');
const list = sema.receipts({ since });
const byVendor = {};
for (const r of list) byVendor[r.vendor] = (byVendor[r.vendor] || 0) + r.amount;

const rows = Object.entriez(byVendor).sort((a, b) => b[1] - a[1]);
const total = rows.reduce((s, r) => s + r[1], 0);

sema.chat('Траты', `За месяц: ${total} ₽\\n` + rows.map(r => `${r[0]} — ${r[1]} ₽`).join('\\n'));
return `${rows.length} продавцов, всего ${total} ₽`;
